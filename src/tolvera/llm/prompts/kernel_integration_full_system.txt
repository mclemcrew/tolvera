Given:

```python
@ti.kernel
def apply_all_experts():
    dt = 0.016  # ~60fps timestep
    
    for i in range(tv.pn):
        if tv.p.field[i].active > 0:
            # Get particle state
            pos = tv.p.field[i].pos
            vel = tv.p.field[i].vel
            mass = tv.p.field[i].mass
            
            # SECTION 1: FORCE ACCUMULATION
            total_force = ti.Vector([0.0, 0.0])
            
            # Single-particle forces
            total_force += expert_gravity(pos, vel, mass) * 1.0
            total_force += expert_attract(pos, vel, mass) * 0.5
            
            # SECTION 2: INTERACTION FORCES
            # Interaction forces (return value from interaction experts)
            total_force += expert_collision_detector(i) * 1.0
            
            # SECTION 3: PHYSICS UPDATE
            # Apply forces
            if mass > 0:
                acceleration = total_force / mass
                tv.p.field[i].vel += acceleration * dt
            
            # Apply damping (0.99 for subtle, 0.98 for normal, 0.95 for strong)
            tv.p.field[i].vel *= 0.99
            
            # Update position
            tv.p.field[i].pos += tv.p.field[i].vel * dt
            
            # SECTION 4: BOUNDARY HANDLING
            # Wrap particles around screen
            if tv.p.field[i].pos[0] > tv.x:
                tv.p.field[i].pos[0] = 0
            elif tv.p.field[i].pos[0] < 0:
                tv.p.field[i].pos[0] = tv.x
                
            if tv.p.field[i].pos[1] > tv.y:
                tv.p.field[i].pos[1] = 0
            elif tv.p.field[i].pos[1] < 0:
                tv.p.field[i].pos[1] = tv.y
            
            # SECTION 5: STATE-BASED CHANGES
            # Only include if states need to trigger changes
            if tv.s.collision_states.field[i].is_collided == 1:
                if tv.p.field[i].species == 0:  # Was blue
                    tv.p.field[i].species = 2    # Becomes orange
```
